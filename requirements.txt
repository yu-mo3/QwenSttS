# QwenSttS 依赖清单
# 安装：.venv\Scripts\python.exe -m pip install -r requirements.txt
#
# Python 版本要求：3.11 ~ 3.12
#   ⚠️ 不要用 3.13+：vosk / webrtcvad / pyaudio 等尚无对应 wheel

# ---- L1 基础层 ----
python-dotenv>=1.0.1        # .env 配置加载

# ---- L2 音频设备层 ----
sounddevice>=0.5.1          # 麦克风采集 / 扬声器播放（PortAudio 绑定，低延迟可中断）
numpy>=1.26.4               # PCM 帧运算、能量计算
pydub>=0.25.1               # 音频格式转换（依赖系统 ffmpeg）

# ---- L4 能力层（云端调用）----
requests>=2.32.3            # 百炼原生端点 / 天气 API
# 实测在 3.28.0 上跑通（依赖 httpx2，注意与 httpx 的日志名不同），
# 加下界以免未来大版本改动流式 chunk 结构
openai>=3.28,<4            # 百炼 OpenAI 兼容端点（LLM / VLM）

# ---- L3 感知层（M3）----
# 注意：webrtcvad-wheels 的导入名仍是 webrtcvad
sherpa-onnx>=1.13.8,<2      # 关键词唤醒（KWS）+ 模型文件
webrtcvad-wheels>=2.0.14    # 语音活动检测（VAD）

# ---- L6 表现层（M4）----
# 用 Essentials 而不是完整 PySide6：完整包会额外拖进 PySide6-WebEngine
# （实测 131.7MB）与 PySide6-Addons（40.5MB），我们一行都用不到。
# Essentials 提供同一个 PySide6 命名空间，只是少了那些用不上的模块。
# 注意：6.12 起 wheel **不再自带 ICU**，conda 环境里的旧 ICU 会抢走
# Qt6Core 需要的 ucnv_* 符号 —— 见 app/ui/qt_runtime.py 的说明与修法。
PySide6-Essentials>=6.8.0

# ---- 视觉（M5 里程碑启用）----
# opencv-python>=4.10.0
